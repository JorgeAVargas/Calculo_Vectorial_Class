<script>

    const explainBtn = document.getElementById('explain-btn');

    const generateBtn = document.getElementById('generate-btn');

    const promptTextarea = document.getElementById('ai-prompt');

    const aiResponseDiv = document.getElementById('aiResponse');

    const loadingDiv = document.getElementById('ai-loading');


    const API_KEY = "";

    const API_URL = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-05-20:generateContent?key=${API_KEY}`;

    

    // Simple markdown to HTML converter

    function simpleMarkdownToHtml(text) {

        // Bold **text**

        text = text.replace(/\*\*(.*?)\*\*/g, '<strong>$1</strong>');

        // Headers ### text

        text = text.replace(/^### (.*$)/gim, '<h3>$1</h3>');

        // List items * text

        text = text.replace(/^\* (.*$)/gim, '<li>$1</li>');

        text = text.replace(/(\<\/li\>)?\n?<li>/g, '<li>');

        text = text.replace(/(<li>.*<\/li>)/gs, '<ul>$1</ul>');

        // Paragraphs (simple version)

        text = text.split('\n').map(p => p.trim() ? `<p>${p}</p>` : '').join('');

        text = text.replace(/<p><\/p>/g, ''); // Clean empty paragraphs

        // Fix lists inside paragraphs

        text = text.replace(/<p><ul>/g, '<ul>').replace(/<\/ul><\/p>/g, '</ul>');

        

        return text;

    }


    async function callGeminiAPI(prompt, maxRetries = 3) {

        loadingDiv.classList.remove('hidden');

        aiResponseDiv.innerHTML = '';

        

        const payload = {

            contents: [{ parts: [{ text: prompt }] }],

        };


        for (let attempt = 0; attempt < maxRetries; attempt++) {

            try {

                const response = await fetch(API_URL, {

                    method: 'POST',

                    headers: { 'Content-Type': 'application/json' },

                    body: JSON.stringify(payload),

                });


                if (!response.ok) {

                    throw new Error(`HTTP error! status: ${response.status}`);

                }


                const result = await response.json();

                

                loadingDiv.classList.add('hidden');

                

                if (result.candidates && result.candidates[0].content && result.candidates[0].content.parts[0].text) {

                    const rawText = result.candidates[0].content.parts[0].text;

                    aiResponseDiv.innerHTML = simpleMarkdownToHtml(rawText);

                } else {

                    aiResponseDiv.innerHTML = '<p class="text-red-500">No se pudo obtener una respuesta válida de la IA. Por favor, intenta de nuevo.</p>';

                }

                return; // Exit after successful call

            } catch (error) {

                console.error('Error calling Gemini API:', error);

                if (attempt === maxRetries - 1) {

                    loadingDiv.classList.add('hidden');

                    aiResponseDiv.innerHTML = `<p class="text-red-500">Error al comunicarse con el Tutor IA después de varios intentos. Revisa la consola para más detalles.</p>`;

                } else {

                    // Exponential backoff

                    const delay = Math.pow(2, attempt) * 1000;

                    await new Promise(res => setTimeout(res, delay));

                }

            }

        }

    }


    explainBtn.addEventListener('click', () => {

        const userInput = promptTextarea.value;

        if (!userInput.trim()) {

            aiResponseDiv.innerHTML = '<p class="text-yellow-600">Por favor, escribe una pregunta o concepto en el área de texto.</p>';

            return;

        }

        const prompt = `Eres un profesor de cálculo multivariable experto y amigable. Explica el concepto clave detrás de la siguiente pregunta o tema de forma clara, concisa y paso a paso para un estudiante universitario. Usa ejemplos simples si es necesario. Pregunta/Tema: "${userInput}"`;

        callGeminiAPI(prompt);

    });


    generateBtn.addEventListener('click', () => {

        const userInput = promptTextarea.value;

        if (!userInput.trim()) {

            aiResponseDiv.innerHTML = '<p class="text-yellow-600">Por favor, escribe una pregunta o concepto en el área de texto para generar un problema similar.</p>';

            return;

        }

        const prompt = `Eres un profesor de cálculo multivariable creando un examen. Genera un nuevo problema de práctica que sea similar en concepto y dificultad a la siguiente pregunta, pero con diferentes números y contexto. Proporciona solo el nuevo problema, 4 opciones de respuesta (una correcta) y la solución explicada paso a paso. Pregunta Original: "${userInput}"`;

        callGeminiAPI(prompt);

    });
